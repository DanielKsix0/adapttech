function addProduct(){
  const name=val('pn').trim(),cost=num(val('pk')),price=num(val('pp')),qty=parseInt(val('pq'))||0,min=parseInt(val('pm'))||0;
  if(!name)return toast('Informe o nome do produto');
  if(qty<1)return toast('Informe a quantidade comprada');
  const p={id:Date.now(),name,cat:val('pc'),cost,price,stock:qty,min,date:val('pd')||today()};
  db.products.push(p);db.buys.push({id:p.id,date:p.date,desc:'Compra: '+name,value:cost*qty});
  save();toast('Produto salvo');go('produtos');
}
function restock(id){
  const p=db.products.find(x=>x.id===id),q=parseInt(prompt('Quantas unidades de "'+p.name+'" entraram?','1'))||0;
  if(q<1)return;p.stock+=q;db.buys.push({id:Date.now(),date:today(),desc:'Reposição: '+p.name,value:p.cost*q});save();toast('Estoque atualizado');render();
}
function delProduct(id){if(!confirm('Excluir este produto? As vendas já registradas continuam nos relatórios.'))return;db.products=db.products.filter(p=>p.id!==id);save();render()}
function addSale(){
  const p=db.products.find(x=>x.id==val('vp')),q=parseInt(val('vq'))||0;
  if(!p||q<1)return toast('Informe a quantidade');
  if(q>p.stock)return toast('Estoque insuficiente: só há '+p.stock);
  db.sales.push({id:Date.now(),pid:p.id,name:p.name,qty:q,price:num(val('vv')),cost:p.cost,fee:num(val('vf')),date:val('vd')||today()});
  p.stock-=q;save();toast('Venda registrada');render();
}

}
